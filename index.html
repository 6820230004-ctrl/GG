const status = document.getElementById("status"); function startGPS() { if (!navigator.geolocation) { status.textContent = "อุปกรณ์ไม่รองรับ GPS"; return; } status.textContent = "กำลังค้นหาตำแหน่ง GPS..."; const options = { enableHighAccuracy: true, timeout: 20000, maximumAge: 0 }; navigator.geolocation.getCurrentPosition( function(position) { const lat = position.coords.latitude; const lng = position.coords.longitude; const accuracy = position.coords.accuracy; status.textContent = `พบตำแหน่ง GPS แล้ว! Latitude: ${lat.toFixed(6)} Longitude: ${lng.toFixed(6)} ความแม่นยำ: ±${Math.round(accuracy)} เมตร`; console.log("GPS:", lat, lng); }, function(error) { let message = ""; switch (error.code) { case 1: message = "กรุณาอนุญาตให้เว็บไซต์ใช้ตำแหน่ง"; break; case 2: message = "ไม่พบสัญญาณ GPS กรุณาเปิด Location"; break; case 3: message = "GPS ใช้เวลานานเกินไป กรุณาลองใหม่"; break; default: message = "เกิดข้อผิดพลาด GPS"; } status.textContent = message; }, options ); } startGPS();
